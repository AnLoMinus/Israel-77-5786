📖 מזמורי תפילה ליום העצמאות

שלושה מזמורים מיוחדים שנכתבו לכבוד יום העצמאות ה-77, משלבים בתוכם שירה, תפילה וחזון לעתיד האומה.

מִזְמוֹר הַשַּׁחַר הַזּוֹהֵר

אוֹר מִן הַמַּרְאֶה

מִן הַשַּׁחַר יַעֲלֶה אוֹר עֲתִיק, מֵעַל הֶהָרִים יָפְרַח שַׁבְתָּךְ.
כְּמָגוֵן אֵשׁ נוֹצֵץ בַּשְּׁמֵי רוּם, כָּבוֹד ה' עַל יְרוּשָׁלַיִם נִגְלָה.

מִכְּנַפֵּי עֲנָנִים יִבְקַע זִיו הַשֵּׁם, וּכְנֶסֶךָ שֶׁמֶן עַל רֹאשׁ עַמּוֹ יִתְּנֶה.
צָבְאוּ הַרְרֵי קֹדֶשׁ לְהַלֵּל אֱלוֹהִים, וּמַעֲיָן נְבוּאָה זָרַם בִּנְאוֹתֵיהֶם.

עַם נוֹסֵעַ בְּדֶרֶךְ תִּקְוָה, עִם דֶּגֶל אֱמוּנָה וְשִׁיר חֲלוֹמוֹת.
וְאֵל שַׁדַּי עֲלֵיהֶם יַזְרִיחַ אֶת אוֹרוֹ, וְיִטַּע שָׁלוֹם בִּגְבוּלָם עַד עֵד.

אֲרוּרוֹת נָפְלוּ בַּחֶשְׁכָה, וְהַשֶּׁמֶשׁ שָׁבָה לְהַאִיר נְשָׁמוֹת.
שַׁעֲרֵי הָאָרֶץ נִפְתְּחוּ לָאֱמוּנָה, וְרוּחַ הַמַּשִּׁיחַ מְרַחֶפֶת בְּתוֹכָהּ.

אֱלֹהֵי הַצְּבָאוֹת, יִהְיוּ שְׁלֵם עִירֶךָ וּשְׁלוּם עַמְּךָ לָנֶצַח.
וּכְבוֹדְךָ יֵרָאֶה עַל כָּל תְּפִלָּה, וּמַלְכוּתְךָ תִּתְעַנֵּג בְּצִיּוֹן שׁוּב.

מִזְמוֹר שִׁבְעִים וְשֶׁבַע

תְּהִלָּה לַתְּקוּמָה

שִׁיר לְיוֹם הָאוֹר, שָׁנָה שֶׁבַע וְשֶׁבַע,
בְּקוֹם עָפָר הִרְאִיתָנוּ אוֹר, מֵאֵפֶר – נִשְׁמָתֵנוּ תְּהִלָּה.

ה' גּוֹאֵל עַמּוֹ מִצָּרוֹת, מֵקִים דּוֹר בְּאֶרֶץ קֹדֶשׁ,
קִיבַּצְתָּ נִדָּחִים בְּחֶסֶד עַתִּיק, בָּנִיתָ נְוֵה תְּפִלָּה מֵאֵין.

מִצָּפוֹן וְעַד דָּרוֹם יִשָּׁמַע תְּרוּעָה, שָׁלוֹם בַּעֲרֵי שָׁלוֹם,
מִן הַיָּם וְעַד מִזְרָח – תִּפְרַח הָאָרֶץ, גְּבוּל יִשְׂרָאֵל יֵרוֹם.

לֹא עַל חֶרֶב נִבְטַח, לֹא בְּרֶכֶב נַעֲלֶה,
בְּשֵׁם ה' נִזְכִּיר, בִּבְרִית עוֹלָם נַחְזִיק.

דָּם קְדוֹשִׁים זוֹרֵחַ מִן הָאֲדָמָה, דִּמְעַת הַשְּׁכִינָה תִּבְנֶה,
בְּקוֹל שִׁירֵנוּ תִּבָּנֶה צִיּוֹן – וְעַל לֵב נִפְעָם תִּשְׁכּוֹן.

תֵּן שָׁלוֹם בַּמְּדִינָה, אוֹר בַּחֲצוֹתֶיהָ,
תְּנֵהֵג שָׂרֶיהָ בְּצֶדֶק, מַנְהִיגֶיהָ בִּתְבוּנָה.

הַלְלוּיָהּ מִשָּׁמַיִם, בִּקְהַל קְדוֹשִׁים,
עַם יִשְׂרָאֵל חַי, וּבְאֶרֶץ תִּהְיֶה תִּקְוָתוֹ.

בָּרוּךְ אַתָּה ה' גּוֹאֵל יִשְׂרָאֵל,
תִּשְׁכּוֹן תְּהִלָּה בִּירוּשָׁלַיִם, וּמִמִּזְבֵּחֲךָ – שָׁלוֹם עֲדֵי עַד.

מזמור לתשפ"ו

שבעים ושבע שנות תחייה

שִׁירוּ לַה' שִׁיר חָדָשׁ, תְּהִלָּתוֹ בִּקְהַל יִשְׂרָאֵל.
יְשֻׁבוּ גְּאוּלִים לְצִיּוֹן בִּרְנָנָה, בִּתְרוּעַ יֹשְׁבֵי אֶרֶץ קֹדֶשׁ יָרִיעוּ.

גָּדַל ה' וּמְהֻלָּל מְאֹד, בְּכָל דּוֹרוֹת שָׁמַר בְּרִיתוֹ.
בִּנְפֹל עַמּוֹ בַּגָּלוּת לֹא עָזָב, וּבְקַצֵּה כָּל אֶרֶץ שָׁמַע נַאֲקָתָם.

קִבֵּץ נִדָּחֵי יִשְׂרָאֵל בְּחֶסֶד, הֵשִׁיב שְׁבוּת צִיּוֹן כַּחֲלוֹם תַּעֲלוּמָה.
עָבְרוּ שִׁבְעִים שָׁנָה וְתוֹסֶפֶת שֶׁבַע, וְכָל נְשָׁמָה תְהַלֵּל אֵל גּוֹאֲלָה.

בָּנוּ גְּדֵרוֹתֵיהֶם בִּדְמָעוֹת וּבִתְפִלּוֹת, וּבַחֲלוֹם שֶׁל דּוֹרוֹת הִשְׁקִימוּ לִפְעֹל.
עַל הָרֵאשִׁים שֶׁל נְפִילֵי צָה"ל נָשְׂאוּ תְּהִלָּה, וְעַל דִּמְעוֹת הַשְּׁכִינָה בָּנָה עַם סִפּוֹר.

הַלְלוּיָהּ לְאֵל יִשְׂרָאֵל, שׁוֹמֵר בְּרִית וְחֶסֶד,
שֶׁמֵּבִיא גְּאֻלָּה בְּאֵמוּנָה, וּמֵעַל לַטֶּבַע.
עֲלוּ בְרִנָּה אֶל צִיּוֹן, נְשְׂאוּ דֶּגֶל הַתְּקוּמָה –
שֶׁבַע וְשִׁבְעִים שָׁנָה לִנְס הַמְּדִינָה!

מִזְמוֹר הַגְּאֻלָּה וְהַתְּקוּמָה

שִׁיר לְיוֹם הַחֵרוּת

קוּמִי אוֹרִי כִּי בָא אוֹרֵךְ, וּכְבוֹד ה' עָלַיִךְ זָרָח.
הִנֵּה הַחֹשֶׁךְ יְכַסֶּה אֶרֶץ, וַעֲרָפֶל לְאֻמִּים,
וְעָלַיִךְ יִזְרַח ה', וּכְבוֹדוֹ עָלַיִךְ יֵרָאֶה.

וְהָלְכוּ גּוֹיִם לְאוֹרֵךְ, וּמְלָכִים לְנֹגַהּ זַרְחֵךְ.
שְׂאִי סָבִיב עֵינַיִךְ וּרְאִי, כֻּלָּם נִקְבְּצוּ בָאוּ לָךְ,
בָּנַיִךְ מֵרָחוֹק יָבֹאוּ, וּבְנוֹתַיִךְ עַל צַד תֵּאָמַנָה.

אָז תִּרְאִי וְנָהַרְתְּ, וּפָחַד וְרָחַב לְבָבֵךְ,
כִּי יֵהָפֵךְ עָלַיִךְ הֲמוֹן יָם, חֵיל גּוֹיִם יָבֹאוּ לָךְ.
שְׂרָפִים יְשַׁבְּחוּךְ, וּמַלְאֲכֵי שָׁלוֹם יְבַכְּרוּךְ.

פִּתְחִי שַׁעֲרַיִךְ תָּמִיד, יוֹמָם וָלַיְלָה לֹא יִסָּגֵרוּ,
לְהָבִיא אֵלַיִךְ חֵיל גּוֹיִם, וּמַלְכֵיהֶם נְהוּגִים.
כִּי הַגּוֹי וְהַמַּמְלָכָה אֲשֶׁר לֹא יַעַבְדוּךְ יֹאבֵדוּ,
וְהַגּוֹיִם חָרֹב יֶחֱרָבוּ.

הַלְלוּיָהּ לְאֵל עֶלְיוֹן, מֶלֶךְ מַלְכֵי הַמְּלָכִים,
שֶׁהֵבִיא גְּאֻלָּה לְעַמּוֹ, וּפְדוּת לְנַחֲלָתוֹ.
יִהְיוּ לְרָצוֹן אִמְרֵי פִינוּ, וְהֶגְיוֹן לִבֵּנוּ לְפָנֶיךָ ה' צוּרֵנוּ וְגֹאֲלֵנוּ.

מִזְמוֹר הָאַחְדוּת

שִׁיר לְאַחִים גַּם יַחְדָּו

הִנֵּה מַה טּוֹב וּמַה נָּעִים, שֶׁבֶת אַחִים גַּם יַחַד.
כְּשֶׁמֶן הַטּוֹב עַל הָרֹאשׁ, יֹרֵד עַל הַזָּקָן זְקַן אַהֲרֹן.

כְּטַל חֶרְמוֹן שֶׁיֹּרֵד עַל הַרְרֵי צִיּוֹן,
כִּי שָׁם צִוָּה ה' אֶת הַבְּרָכָה, חַיִּים עַד הָעוֹלָם.

אַחִים אֲנַחְנוּ, בְּנֵי אָב אֶחָד,
בְּלֵב אֶחָד נְהַלֵּךְ בְּדֶרֶךְ אֱמוּנָה.
לֹא יִשָּׂא גּוֹי אֶל גּוֹי חֶרֶב,
וְלֹא יִלְמְדוּ עוֹד מִלְחָמָה.

בְּשָׁלוֹם יַחְדָּו אֶשְׁכְּבָה וְאִישָׁן,
כִּי אַתָּה ה' לְבָדָד לָבֶטַח תּוֹשִׁיבֵנִי.
וְעַל כַּנְפֵי נְשָׁרִים יִשָּׂאוּנִי,
וְאֶל הַר קָדְשְׁךָ יְבִיאוּנִי.

בָּרוּךְ אַתָּה ה' מְיַחֵד עַמּוֹ יִשְׂרָאֵל בְּאַהֲבָה,
מְקַבֵּץ נִדְחֵי עַמּוֹ יִשְׂרָאֵל מֵאַרְבַּע כַּנְפוֹת הָאָרֶץ.
יִהְיוּ לְרָצוֹן אִמְרֵי פִינוּ וְהֶגְיוֹן לִבֵּנוּ לְפָנֶיךָ ה' צוּרֵנוּ וְגֹאֲלֵנוּ.

שִׁיר הַמַּעֲלוֹת לִירוּשָׁלַיִם

מִזְמוֹר לְצִיּוֹן הַמְּקֻדֶּשֶׁת

שִׁיר הַמַּעֲלוֹת, בְּשׁוּב ה' אֶת שִׁיבַת צִיּוֹן,
הָיִינוּ כְּחֹלְמִים, כְּמוֹ חֲלוֹם תַּעֲלוּמָה.
אָז יִמָּלֵא שְׂחוֹק פִּינוּ, וּלְשׁוֹנֵנוּ רִנָּה,
אָז יֹאמְרוּ בַגּוֹיִם, הִגְדִּיל ה' לַעֲשׂוֹת עִם אֵלֶּה.

יְרוּשָׁלַיִם הַבְּנוּיָה כְּעִיר שֶׁחֻבְּרָה לָּהּ יַחְדָּו,
שָׁם עָלוּ שְׁבָטִים שִׁבְטֵי יָהּ, עֵדוּת לְיִשְׂרָאֵל.
לְהֹדוֹת לְשֵׁם ה', כִּי שָׁם יָשְׁבוּ כִסְאוֹת לְמִשְׁפָּט,
כִּסְאוֹת לְבֵית דָּוִד.

שַׁאֲלוּ שְׁלוֹם יְרוּשָׁלַיִם, יִשְׁלָיוּ אֹהֲבָיִךְ,
יְהִי שָׁלוֹם בְּחֵילֵךְ, שַׁלְוָה בְּאַרְמְנוֹתָיִךְ.
לְמַעַן אַחַי וְרֵעָי, אֲדַבְּרָה נָּא שָׁלוֹם בָּךְ,
לְמַעַן בֵּית ה' אֱלֹהֵינוּ, אֲבַקְשָׁה טוֹב לָךְ.

הִנֵּה אֲנַחְנוּ בָּאנוּ אֶל הַמָּקוֹם אֲשֶׁר אָמַר ה',
זֹאת מְנוּחָתִי עֲדֵי עַד, פֹּה אֵשֵׁב כִּי אִוִּיתִיהָ.
וְעַל הַר צִיּוֹן שֶׁשָּׁמֵם, שׁוּעָלִים הִלְּכוּ בוֹ,
תָּקוּם וְתִבָּנֶה עִיר הָאֱלֹהִים.

בָּרוּךְ אַתָּה ה' בּוֹנֵה יְרוּשָׁלַיִם,
מְקַבֵּץ נִדְחֵי עַמּוֹ יִשְׂרָאֵל מֵאַרְבַּע כַּנְפוֹת הָאָרֶץ.
יִהְיוּ לְרָצוֹן אִמְרֵי פִינוּ וְהֶגְיוֹן לִבֵּנוּ לְפָנֶיךָ ה' צוּרֵנוּ וְגֹאֲלֵנוּ.

מִזְמוֹר הַשָּׁלוֹם וְהַבְּרָכָה

תְּפִלָּה לְשָׁלוֹם הָאָרֶץ

ה' יְבָרֵךְ אֶת עַמּוֹ בַשָּׁלוֹם,
וְיִתֵּן לָנוּ שָׁלוֹם מִכָּל עֲבָרָיו.
יְבָרֵךְ אֶת בָּתֵּינוּ וְאֶת שָׂדוֹתֵינוּ,
וְיִתֵּן שָׁלוֹם בָּאָרֶץ.

יְבָרֵךְ אֶת חֵילֵנוּ וְאֶת מַנְהִיגֵינוּ,
וְיִתֵּן בְּלִבָּם חָכְמָה וּתְבוּנָה.
יְבָרֵךְ אֶת יוֹצְאֵי צְבָאֵנוּ וְאֶת בָּאֵי שַׁעֲרֵינוּ,
וְיִתֵּן שָׁלוֹם בַּגְּבוּלוֹת.

יְבָרֵךְ אֶת עוֹלָמֵינוּ וְאֶת תַּלְמִידֵינוּ,
וְיִתֵּן בָּהֶם רוּחַ חָכְמָה וּבִינָה.
יְבָרֵךְ אֶת עֲבוֹדָתֵינוּ וְאֶת מַעֲשֵׂה יָדֵינוּ,
וְיִתֵּן שָׁלוֹם בַּמְּדִינָה.

יְבָרֵךְ אֶת כָּל יוֹשְׁבֵי תֵבֵל,
וְיִתֵּן שָׁלוֹם בָּעוֹלָם כֻּלּוֹ.
יְבָרֵךְ אֶת כָּל בְּנֵי הָאָדָם,
וְיִתֵּן שָׁלוֹם בֵּין הָאֻמּוֹת.

בָּרוּךְ אַתָּה ה' מְבָרֵךְ עַמּוֹ יִשְׂרָאֵל בַּשָּׁלוֹם,
מֵבִיא גְּאֻלָּה לְעַמּוֹ יִשְׂרָאֵל בְּרַחֲמִים.
יִהְיוּ לְרָצוֹן אִמְרֵי פִינוּ וְהֶגְיוֹן לִבֵּנוּ לְפָנֶיךָ ה' צוּרֵנוּ וְגֹאֲלֵנוּ.

מִזְמוֹר הַתְּקוּמָה וְהַתְּחִיָּה

שִׁיר לְדוֹר הַתְּקוּמָה

מֵעָפָר קַמְתִּי, מֵאֵפֶר נוֹעַרְתִּי,
כְּעֵץ שֶׁנִּטַּע עַל פַּלְגֵי מַיִם.
מִן הַגּוֹלָה הָעֲצוּמָה, מִן הַשְּׁאוֹל הַתַּחְתִּית,
הֵעִיר ה' אֶת רוּחִי, וַיְחַיֵּנִי מִן הַמֵּתִים.

הִנֵּה אֲנִי חַי, הִנֵּה אֲנִי קַיָּם,
כְּעֵץ פֹּרִיָּה בְּאֶרֶץ נְוֵה.
מִן הַחֹרֶב הָעֲצוּם, מִן הַשְּׁמָמָה הַגְּדוֹלָה,
הֵקִיץ ה' אֶת נַפְשִׁי, וַיְחַיֵּנִי מִן הַקֶּבֶר.

בָּנִיתִי בָּתִּים וְנָטַעְתִּי כְּרָמִים,
הֵקַמְתִּי עָרִים וְיִשַּׁבְתִּי בָּהֶן.
מִן הַחֻרְבָּן הָעַתִּיק, מִן הַשְּׁמָמָה הַיְשָׁנָה,
הֵעִיר ה' אֶת רוּחִי, וַיְחַיֵּנִי מִן הַמֵּתִים.

הִנֵּה אֲנִי חַי, הִנֵּה אֲנִי קַיָּם,
כְּעֵץ פֹּרִיָּה בְּאֶרֶץ נְוֵה.
מִן הַחֹרֶב הָעֲצוּם, מִן הַשְּׁמָמָה הַגְּדוֹלָה,
הֵקִיץ ה' אֶת נַפְשִׁי, וַיְחַיֵּנִי מִן הַקֶּבֶר.

בָּרוּךְ אַתָּה ה' מְחַיֵּה הַמֵּתִים,
מֵבִיא גְּאֻלָּה לְעַמּוֹ יִשְׂרָאֵל בְּרַחֲמִים.
יִהְיוּ לְרָצוֹן אִמְרֵי פִינוּ וְהֶגְיוֹן לִבֵּנוּ לְפָנֶיךָ ה' צוּרֵנוּ וְגֹאֲלֵנוּ.

מִזְמוֹר הַחֲלוֹם וְהַתִּקְוָה

שִׁיר לְעוֹלָם הַבָּא

בַּיָּמִים הָהֵם וּבָעֵת הַהִיא,
אָשִׁיב אֶת שְׁבוּת יְהוּדָה וִירוּשָׁלַיִם.
וְקִבַּצְתִּי אֶת כָּל הַגּוֹיִם,
וְהוֹרַדְתִּים אֶל עֵמֶק יְהוֹשָׁפָט.

וְשָׁפַטְתִּי אֶת כָּל הַגּוֹיִם מִסָּבִיב,
עַל עַמִּי וְנַחֲלָתִי יִשְׂרָאֵל.
אֲשֶׁר פִּזְּרוּ בַגּוֹיִם,
וְאֶת אַרְצִי חִלֵּקוּ.

וְעַל עַמִּי יַדּוּ גוֹרָל,
וַיִּתְּנוּ הַיֶּלֶד בַּזּוֹנָה,
וְהַיַּלְדָּה מָכְרוּ בַיַּיִן,
וַיִּשְׁתּוּ.

וְגַם מָה אַתֶּם לִי צֹר וְצִידוֹן,
וְכֹל גְּלִילוֹת פְּלָשֶׁת.
הַגְּמוּל אַתֶּם מְשַׁלְּמִים עָלָי,
וְאִם גֹּמְלִים אַתֶּם עָלַי,
קַל מְהֵרָה אָשִׁיב גְּמוּלְכֶם בְּרֹאשְׁכֶם.

בָּרוּךְ אַתָּה ה' מְקַיֵּם הַבְּרִית וְהַחֶסֶד,
מֵבִיא גְּאֻלָּה לְעַמּוֹ יִשְׂרָאֵל בְּרַחֲמִים.
יִהְיוּ לְרָצוֹן אִמְרֵי פִינוּ וְהֶגְיוֹן לִבֵּנוּ לְפָנֶיךָ ה' צוּרֵנוּ וְגֹאֲלֵנוּ.
